    <!-- JavaScript Logical Implementation -->
    <script>
        const chatMessages = document.getElementById('chatMessages');
        const userInput = document.getElementById('userInput');
        const typingIndicator = document.getElementById('typingIndicator');

        // Preset Prompt Auto Fill
        function applyPreset(text) {
            userInput.value = text;
            userInput.focus();
        }

        // Handle Send Form
        function handleSend(event) {
            event.preventDefault();
            const query = userInput.value.trim();
            if (!query) return;

            // 1. Add User Message
            appendUserMessage(query);
            userInput.value = '';

            // 2. Show Typing Indicator
            showTyping(true);

            // 3. Simulate AI Response after 1 second delay
            setTimeout(() => {
                showTyping(false);
                const simulatedAnswer = getSimulatedResponse(query);
                appendAIMessage(simulatedAnswer);
            }, 1000);
        }

        // Render User Message
        function appendUserMessage(text) {
            const msgHTML = `
                <div class="flex items-start justify-end space-x-3 max-w-[85%] sm:max-w-[75%] ml-auto">
                    <div class="bg-emerald-600 p-3.5 rounded-2xl rounded-tr-none text-white text-sm leading-relaxed shadow-sm">
                        ${escapeHtml(text)}
                    </div>
                    <div class="w-8 h-8 rounded-full bg-slate-700 flex-shrink-0 flex items-center justify-center text-white text-xs font-bold shadow">
                        আপনি
                    </div>
                </div>
            `;
            chatMessages.insertAdjacentHTML('beforeend', msgHTML);
            scrollToBottom();
        }

        // Render AI Message
        function appendAIMessage(text) {
            const msgHTML = `
                <div class="flex items-start space-x-3 max-w-[85%] sm:max-w-[75%]">
                    <div class="w-8 h-8 rounded-full bg-emerald-600 flex-shrink-0 flex items-center justify-center text-white text-xs font-bold shadow">
                        AI
                    </div>
                    <div class="bg-white p-3.5 rounded-2xl rounded-tl-none border border-slate-200 shadow-sm text-slate-800 text-sm leading-relaxed">
                        ${text}
                    </div>
                </div>
            `;
            chatMessages.insertAdjacentHTML('beforeend', msgHTML);
            scrollToBottom();
        }

        // Typing Indicator Toggle
        function showTyping(show) {
            if (show) {
                typingIndicator.classList.remove('hidden');
            } else {
                typingIndicator.classList.add('hidden');
            }
            scrollToBottom();
        }

        // Scroll to Bottom Helper
        function scrollToBottom() {
            chatMessages.scrollTop = chatMessages.scrollHeight;
        }

        // Reset Chat
        function resetChat() {
            chatMessages.innerHTML = `
                <div class="flex items-start space-x-3 max-w-[85%] sm:max-w-[75%]">
                    <div class="w-8 h-8 rounded-full bg-emerald-600 flex-shrink-0 flex items-center justify-center text-white text-xs font-bold shadow">
                        AI
                    </div>
                    <div class="bg-white p-3.5 rounded-2xl rounded-tl-none border border-slate-200 shadow-sm text-slate-800 text-sm leading-relaxed">
                        👋 <strong>নতুন চ্যাট শুরু হয়েছে!</strong><br><br>
                        আমি আপনাকে কী বিষয়ে সাহায্য করতে পারি?
                    </div>
                </div>
            `;
            userInput.value = '';
        }

        // Demo Simulated Intelligence Logic
        function getSimulatedResponse(query) {
            const lower = query.toLowerCase();
            
            if (lower.includes('data entry') || lower.includes('ডাটা এন্ট্রি')) {
                return `<strong>Data Entry সংক্রান্ত পরামর্শ (Demo Response):</strong><br><br>
                ১. <strong>Excel & Google Sheets:</strong> Fast Typing এবং Keyboard Shortcuts ভালোমতো শিখুন।<br>
                ২. <strong>Accuracy:</strong> ভুলের পরিমাণ কমাতে ডাটা ভ্যালিডেশন অপশন ব্যবহার করুন।<br>
                ৩. <strong>Tooling:</strong> MS Excel, Google Docs, Canva, Notion এবং OCR টেক্সট কনভার্টার জানা থাকলে কাজ পেতে সুবিধা হবে।`;
            } else if (lower.includes('excel') || lower.includes('vlookup') || lower.includes('সূত্র')) {
                return `<strong>Excel Formula টিপস (Demo Response):</strong><br><br>
                <code>=VLOOKUP(lookup_value, table_array, col_index_num, [range_lookup])</code><br><br>
                উদাহরণ: নির্দিষ্ট কোনো গ্রাহকের ID দিয়ে নাম বা ব্যালেন্স খোঁজার জন্য VLOOKUP খুব দরকারী একটি ফর্মুলা।`;
            } else if (lower.includes('freelance') || lower.includes('ফ্রিল্যান্সিং')) {
                return `<strong>Freelancing সফলতার নির্দেশিকা (Demo Response):</strong><br><br>
                ১. নির্দিষ্ট একটি বিষয়ে Skill অর্জনে অন্তত ৩-৬ মাস সময় দিন।<br>
                ২. ভালো একটি পোরটফোলিও (Portfolio) তৈরি করুন।<br>
                ৩.Upwork, Fiverr বা LinkedIn-এ আবেদন করার সময় ক্লায়েন্টের প্রবলেম সমাধানের ওপর গুরুত্ব দিন।`;
            } else {
                return `ধন্যবাদ আপনার প্রশ্নের জন্য: <em>"${escapeHtml(query)}"</em><br><br>✅ <strong>ইন্টারফেস টেস্ট সফল:</strong> সফটওয়্যারটি আপনার বার্তা সঠিকভাবে গ্রহণ করেছে! এটি বর্তমানে ডেমো মোডে রয়েছে। পরবর্তী ধাপে বাস্তব AI API কানেক্ট করার পর এখানে সরাসরি এআই-এর পূর্ণাঙ্গ উত্তর আসবে।`;
            }
        }

        // Security Helper
        function escapeHtml(text) {
            return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }
